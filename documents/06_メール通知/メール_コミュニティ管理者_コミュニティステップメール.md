# メール_コミュニティ管理者_コミュニティステップメール

## 1. 概要

### 1.1 機能の目的
コミュニティ管理者に対して、コミュニティの活用促進を目的とした段階的なリマインドメールを自動送信する機能です。

### 1.2 主要な機能
- コミュニティ作成後のイベント未作成に対するリマインド
- イベント下書き保存後の公開促進リマインド
- イベント開催後の継続開催促進リマインド
- 重複送信を防ぐステータス管理
- 段階的な送信タイミング制御
- **送信回数に応じたメール内容の動的変更**

### 1.3 対象ユーザー
- コミュニティ管理者全員
- 複数コミュニティを管理している管理者

### 1.4 メール内容案
#### コミュニティ作成後に放置している場合
✅ 対象者：コミュニティ作成後、7日以内にイベント未作成

📅 タイミング：作成後 3日後、2週間後、1ヶ月後、3ヶ月後、6ヶ月後、9ヶ月後、12ヶ月後、18ヶ月、24ヶ月のタイミングで送信

📩 メール内容案：
件名：「はじめの一歩を踏み出してみませんか？🍽」

#### イベント下書きのまま放置している場合
✅ 対象者：イベント下書き保存から48時間以内に公開されていない

📅 タイミング：下書き保存から、3日後、2週間後、1ヶ月後、3ヶ月後、6ヶ月後、9ヶ月後、12ヶ月後、18ヶ月、24ヶ月のタイミングで送信

📩 メール内容案：
件名：「あとワンクリックでイベントが公開されます！🎉」

#### イベントを1回開催したが、2回目以降が開催されていない場合
✅ 対象者：最終開催日からXX日間、新しいイベントが作成されていない

📅 タイミング：前回イベントから 3日後、2週間後、1ヶ月後、3ヶ月後、6ヶ月後、9ヶ月後、12ヶ月後、18ヶ月、24ヶ月のタイミングで送信

📩 メール内容案：
件名：「また、あの時間をつくりませんか？」

### 1.5 リマインドステータス管理
| 状況         　　　　 | リマインドのステータス  | 送信タイミング                            　               　         | 送信内容             |
| ------------------- | ------------------- | ------------------------------------------------------------------- | ------------------ |
| コミュニティ作成のみ   | `community_created` | 3日後、2週間後、1ヶ月後、3ヶ月後、6ヶ月後、9ヶ月後、12ヶ月後、18ヶ月、24ヶ月 | イベント作成リマインド |

#### 5.1.3 Zodスキーマ定義
```typescript
// common/src/schemas/community.ts
import { z } from 'zod';

export const CommunityDbSchema = z.object({
  // 既存フィールド...
  
  // 新規追加フィールド
  community_remind_step: z.enum(['community_created', 'event_draft', 'event_repeat']).nullable(),
  community_remind_last_sent: z.date().nullable(),
  community_remind_next_send: z.date().nullable(),
  community_remind_sent_count: z.number().min(0).default(0),
});

export class Community {
  // 既存フィールド...
  
  // 新規追加フィールド
  community_remind_step: 'community_created' | 'event_draft' | 'event_repeat' | null = null;
  community_remind_last_sent: Date | null = null;
  community_remind_next_send: Date | null = null;
  community_remind_sent_count: number = 0;
  
  // バリデーション
  validateRemindStep(): boolean {
    return this.community_remind_step === null || 
           ['community_created', 'event_draft', 'event_repeat'].includes(this.community_remind_step);
  }
  
  // ファクトリメソッド
  static fromDb(data: any): Community {
    const item = new Community();
    // 既存フィールドの設定...
    
    // 新規フィールドの設定
    item.community_remind_step = data.community_remind_step || null;
    item.community_remind_last_sent = data.community_remind_last_sent?.toDate() || null;
    item.community_remind_next_send = data.community_remind_next_send?.toDate() || null;
    item.community_remind_sent_count = data.community_remind_sent_count || 0;
    
    return item;
  }
}
```

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /communities/{communityId} {
      // 既存ルール...
      
      // リマインドフィールドの更新権限
      allow update: if request.auth != null && 
        (resource.data.managers[request.auth.uid] == true || 
         request.auth.token.admin == true) &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['community_remind_step', 'community_remind_last_sent', 
                   'community_remind_next_send', 'community_remind_sent_count']);
    }
  }
}
```

### 5.6 Functions設計

#### 5.6.1 Scheduled Functions（定期実行）
```typescript
// functions/default/src/scheduled/communityStepMail.ts
import * as fs from 'fs';
import * as path from 'path';
import { StepMailTemplates, StepMailTemplate } from '../types/stepMailTemplate';

export const sendCommunityStepMails = onSchedule(
  '0 10 * * *',  // 毎日午前10時
  async () => {
    console.log('Starting community step mail sending...');
    
    try {
      const now = new Date();
      const communities = await getCommunitiesForRemind(now);
      
      for (const community of communities) {
        await sendStepMail(community);
        await updateRemindStatus(community.id);
      }
      
      console.log(`Sent ${communities.length} step mails`);
    } catch (error) {
      console.error('Community step mail failed:', error);
    }
  }
);

// リマインド対象コミュニティの取得
async function getCommunitiesForRemind(now: Date) {
  const snapshot = await db.collection('communities')
    .where('community_remind_step', 'in', ['community_created', 'event_draft', 'event_repeat'])
    .where('community_remind_next_send', '<=', now)
    .get();
    
  return snapshot.docs.map(doc => ({
    id: doc.id,
    ...doc.data()
  }));
}

// ステップメール送信（送信回数に応じたテンプレート選択）
async function sendStepMail(community: any) {
  // 送信回数に応じたテンプレートを取得
  const template = getStepMailTemplate(community.community_remind_step, community.community_remind_sent_count);
  
  if (!template) {
    console.warn(`No template found for step: ${community.community_remind_step}, count: ${community.community_remind_sent_count}`);
    return;
  }
  
  // コミュニティ管理者全員に送信
  const managers = Object.keys(community.managers || {});
  
  for (const managerId of managers) {
    const user = await getUser(managerId);
    if (user?.user_email) {
      await sendGrid.send({
        to: user.user_email,
        from: 'noreply@bokudeli.com',
        subject: template.subject,
        templateId: template.sendgrid_template_id,
        dynamicTemplateData: {
          community_name: community.community_name,
          community_account: community.community_account,
          manager_name: user.user_name || '管理者様',
          send_count: community.community_remind_sent_count + 1,
          step_type: community.community_remind_step
        }
      });
    }
  }
}
```

### 5.8 送信タイミング計算
```typescript
function calculateNextSendDate(step: string, sentCount: number): Date {
  const now = new Date();
  const intervals = [3, 14, 30, 90, 180, 270, 365, 540, 730]; // 日数
  
  if (sentCount >= intervals.length) {
    return null; // 送信終了
  }
  
  const daysToAdd = intervals[sentCount];
  const nextDate = new Date(now);
  nextDate.setDate(nextDate.getDate() + daysToAdd);
  
  return nextDate;
}
```

```typescript
import * as fs from 'fs';

export async function updateTemplateFile(newTemplates: StepMailTemplates) {

export async function validateTemplateFile(templates: StepMailTemplates): Promise<boolean> {
  try {
    // 必須フィールドのチェック
    if (!templates.version || !templates.last_updated || !templates.templates) {
      return false;
    }
    
    // 各ステップタイプのチェック
    const stepTypes = ['community_created', 'event_draft', 'event_repeat'];
    for (const stepType of stepTypes) {
      if (!templates.templates[stepType as keyof typeof templates.templates]) {
        return false;
      }
      
      // 送信回数0-8のチェック
      for (let i = 0; i <= 8; i++) {
        const template = templates.templates[stepType as keyof typeof templates.templates][i.toString()];
        if (!template || !template.subject || !template.content || !template.sendgrid_template_id) {
          return false;
        }
      }
    }
    
    return true;
  } catch (error) {
    console.error('Template validation failed:', error);
    return false;
  }
}
```
